export function checkToolsRegistered(tools: Anthropic.Tool[]): void {
  const expected: Record<string, string> = { calculator: "expression", web_search: "query" };
  const names = tools.map((t) => t.name);
  if (tools.length !== 2 || !names.includes("calculator") || !names.includes("web_search")) {
    throw new Error(`Step 1 (tools registered): expected calculator and web_search, got [${names}]`);
  }
  for (const tool of tools) {
    const field = expected[tool.name];
    if (!tool.description || tool.input_schema.type !== "object" || !tool.input_schema.required?.includes(field)) {
      throw new Error(
        `Step 1 (tools registered): ${tool.name} needs a description and an object input_schema requiring "${field}"`
      );
    }
  }
  logCheck(`tools registered: ${names.join(", ")} (input_schema ok)`);
}

What: TOOLS holds the two tool definitions: a name, a description Claude reads when choosing, and a JSON Schema input_schema. checkToolsRegistered() throws unless both tools exist, each with an object schema and its required field, and logs CHECK tools registered.

Why: Claude never runs a tool; it only picks one by name and writes input that fits the schema. Two tools, not one, make the choice of tool a model decision: that is the model-driven part of the loop. The array is sent with every request.

Why: The Messages API keeps no memory between calls, so this array is the conversation and goes out in full each turn. The cap is a safety net only; it is not how a normal run ends.

5. end_turn is the only normal exit

src/agent-loop.ts:72-83

    // Step 4: end_turn is Claude saying the task is complete. It is the only normal exit.
    if (response.stop_reason === "end_turn") {
      logLoopExit(iteration, MAX_ITERATIONS);
      return { text, iterations: iteration, toolCalls, usage };
    }

    // Any other value (max_tokens, refusal, ...) means "not finished, and not asking for a tool".
    // Each needs its own handling (raise max_tokens, report the refusal); this exercise does not
    // need them, so it stops loudly rather than treat the answer as complete or loop blindly.
    if (response.stop_reason !== "tool_use") {
      throw new Error(`Step 4 (stop_reason): unexpected stop_reason=${response.stop_reason}`);
    }

What: On stop_reason === "end_turn" it logs the exit and returns the text. Any value other than tool_use throws.

Why: This is the lesson's core rule: stop_reason alone decides. No phrase matching, no content[0].type check. An unexpected value such as max_tokens stops loudly instead of being taken as an answer.

      toolResults.push({ type: "tool_result", tool_use_id: block.id, content: out.content, is_error: out.isError });

    // Cannot fail as written; it guards the API rule (one result per tool_use id) against later edits.
    checkToolResultIds(toolUses, toolResults);
export function runTool(name: string, input: unknown): ToolOutput {
  // A missing field arrives as the string "undefined", which the calculator rejects as an error.
  const args = (input ?? {}) as Record<string, unknown>;
  try {
    switch (name) {
      case "calculator":
        return { content: JSON.stringify({ result: evaluate(String(args.expression)) }), isError: false };
      case "web_search":
        return { content: webSearch(String(args.query)), isError: false };
      default:
        return { content: `Unknown tool: ${name}`, isError: true };
    }
  } catch (err) {
    return { content: `Error: ${err instanceof Error ? err.message : err}`, isError: true };
  }
}

What: Each tool_use block goes to runTool(), and its output becomes a tool_result carrying the block's id as tool_use_id. checkToolResultIds() asserts one result per id.

Why: A response can hold several tool_use blocks, and the API rejects a turn that leaves one unanswered. A failing tool returns is_error instead of crashing, so Claude reads the error and can correct its input.

9. The sequential test, checked in code

export function checkSequential(calls: ToolCall[], answer: string): void {
  // Iterations that ran at least one tool: two means two separate round trips to Claude.
  const toolIterations = new Set(calls.map((c) => c.iteration)).size;
  // Calculator calls made after the first search (none if there was no search).
  const firstSearch = calls.findIndex((c) => c.name === "web_search");
  const calcAfterSearch =
    firstSearch < 0 ? [] : calls.slice(firstSearch + 1).filter((c) => c.name === "calculator");
  // Did Claude carry the searched number into its calculation?
  const usedValue = calcAfterSearch.some((c) => {
    const { expression } = c.input as { expression?: unknown };
    return mentionsNumber(String(expression ?? ""), SEARCH_VALUE);
  });
  const order = calls.map((c) => c.name).join(" -> ");

  if (toolIterations < 2) {
    throw new Error(`Step 5 (sequential): only ${toolIterations} tool iteration(s): ${order || "(none)"}`);
  }
  if (calcAfterSearch.length === 0) {
    throw new Error(`Step 5 (sequential): no calculator call after web_search: ${order}`);
  }
  if (!usedValue) {
    throw new Error(`Step 5 (sequential): no calculator expression uses the search value ${SEARCH_VALUE}`);
  }
  logCheck(`sequential: ${toolIterations} tool iterations, ${order}, search value used`);

  // Wording varies run to run; this accepts "560,000" or "560000" but not "0.56 million".
  if (!mentionsNumber(answer, EXPECTED_ANSWER)) {
    throw new Error(`Step 5 (sequential): final answer does not state ${EXPECTED_ANSWER}`);
  }
  logCheck(`final answer states ${EXPECTED_ANSWER}`);
}

What: Asserts at least two tool iterations, a calculator call after the first web_search, the searched value 5600000 in the calculator expression, and 560000 in the final answer.

Why: The test prompt needs a search, then a calculation on its result. Claude has to chain the tools over separate round trips, carrying the value from one tool_result into the next tool_use.
